cmake_minimum_required(VERSION 3.16)
project(Neutrino VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Neutrino is the Windows 10+ package; use NeutrinoOnLinux on Unix.")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(NEUTRINO_BUILD_EXAMPLES "Build the existing demonstration applications" ON)
add_library(Neutrino STATIC
    src/win32/Neu_Win32.cpp
    src/Neu_Cards.cpp
)
add_library(Neutrino::Neutrino ALIAS Neutrino)
target_compile_features(Neutrino PUBLIC cxx_std_17)
target_include_directories(Neutrino PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(Neutrino PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
    _CRT_SECURE_NO_WARNINGS WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(Neutrino PUBLIC user32 gdi32 comctl32)
if(MSVC)
    target_compile_options(Neutrino PUBLIC /utf-8 /EHsc)
endif()
